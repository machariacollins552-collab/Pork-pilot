function chart(){
  var t=sod(),days=[];
  for(var i=6;i>=0;i--){var a=t-i*DAY,r=[a,a+DAY],rev=0,cost=0;
    S.sales.forEach(function(s){if(inR(s.ts,r)){rev+=s.total;cost+=ENG.cogs[s.id]||0}});
    days.push({a:a,rev:rev,gp:Math.max(0,rev-cost)})}
  var max=Math.max.apply(null,days.map(function(d){return d.rev}));
  if(max<=0)return '';
  var step=Math.pow(10,Math.floor(Math.log10(max)));var top=Math.ceil(max/step)*step; if(top/step<2)top=step*2;
  var W=320,H=150,L=34,B=22,T0=10,cw=(W-L)/7,ch=H-B-T0;
  var y=function(v){return T0+ch-(v/top)*ch};
  var s='<svg class="chart" viewBox="0 0 '+W+' '+H+'" role="img" aria-label="Sales and gross profit for the last 7 days" width="100%">';
  [0,top/2,top].forEach(function(v){s+='<line x1="'+L+'" x2="'+W+'" y1="'+y(v)+'" y2="'+y(v)+'" stroke="var(--line)" stroke-width="1"/><text x="'+(L-5)+'" y="'+(y(v)+4)+'" text-anchor="end">'+(v>=1000?(v/1000)+'k':v)+'</text>'});
  days.forEach(function(d,i){var x=L+i*cw+cw/2,bw=cw*0.3;
    s+='<rect x="'+(x-bw-1)+'" y="'+y(d.rev)+'" width="'+bw+'" height="'+Math.max(0,y(0)-y(d.rev))+'" rx="3" fill="var(--accent)"/>';
    s+='<rect x="'+(x+1)+'" y="'+y(d.gp)+'" width="'+bw+'" height="'+Math.max(0,y(0)-y(d.gp))+'" rx="3" fill="var(--pink)"/>';
    var lbl=new Date(d.a).toLocaleDateString('en-KE',{weekday:'narrow'});
    s+='<text x="'+x+'" y="'+(H-6)+'" text-anchor="middle" class="'+(i===6?'tt':'')+'">'+lbl+'</text>';
  });
  s+='</svg>';
  return '<div class="card"><h3>Last 7 days</h3>'+s+'<div class="legend"><span><i style="background:var(--accent)"></i>Sales (KSh)</span><span><i style="background:var(--pink)"></i>Gross profit</span></div></div>';
}

function cartTotal(){return rnd2(sum(cart.items,function(i){return N(i.qty)*N(i.price)}))}
function cartPaid(t){return cart.manual?N(cart.paid):(cart.method==='credit'?0:t)}
function updateCart(){
  if(!cart)return;
  cart.items.forEach(function(it,i){var e=$('#lt'+i);if(e)e.textContent=money(N(it.qty)*N(it.price))});
  var t=cartTotal(),p=cartPaid(t),bal=t-p;
  var e=$('#saleTotal');if(e)e.textContent=money(t);
  var b=$('#saleBal');if(b)b.textContent=bal>0.5?('On credit: '+money(bal)):(p>t+0.5?'Received is more than total':'Fully paid: '+money(p));
  var f=$('#fPaid');if(f&&!cart.manual)f.placeholder=num(p);
}
async function saveSale(){
  if(saving||!cart)return;
  var items=cart.items.filter(function(i){return N(i.qty)>0}).map(function(i){return {pid:i.pid,name:i.name,qty:rnd2(N(i.qty)),price:N(i.price)}});
  if(!items.length)return toast('Add a cut and enter its weight.');
  var total=rnd2(sum(items,function(i){return i.qty*i.price})),paid=rnd2(cartPaid(total));
  if(paid>total+0.5)return toast('Amount received is more than the total.');
  var cid=cart.cid;
  if(cid==='__new'&&!cart.newName.trim())return toast('Enter the new customer’s name.');
  if(total-paid>0.5&&!cid)return toast('Choose a customer so the balance can be tracked.');
  saving=true;var btn=$('#saveSale');if(btn)btn.disabled=true;
  try{
    if(cid==='__new'){var c=await put('customers',{name:cart.newName.trim(),phone:cart.newPhone.trim(),ts:Date.now()},{quiet:true});cid=c.id}
    var ts=Date.now(),s={id:uid(),ts:ts,no:receiptNo(ts),cid:cid||null,items:items,total:total,paid:Math.min(paid,total),method:cart.method,ref:(cart.ref||'').trim().toUpperCase()};
    await put('sales',s);
    var neg=items.filter(function(i){return ENG.P[i.pid]&&ENG.P[i.pid].stock<-0.005}).map(function(i){return i.name});
    receipt(s.id);
    if(neg.length)toast('Stock for '+neg.join(', ')+' is now below zero. Record your stock in.');
  }finally{saving=false}
}
function receiptNo(ts){var d=new Date(ts),k=S.sales.filter(function(s){return sod(s.ts)===sod(ts)}).length+1;return 'PP-'+String(d.getFullYear()).slice(2)+String(d.getMonth()+1).padStart(2,'0')+String(d.getDate()).padStart(2,'0')+'-'+String(k).padStart(3,'0')}

function receiptText(s){
  var st=settings(),c=cust(s.cid),L='';
  L+=st.shop+'\n'+(st.phone?'Tel '+st.phone+'\n':'')+'Receipt '+s.no+'\n'+dts(s.ts)+'\n'+(c?'Customer: '+c.name+'\n':'')+'--------------------------\n';
  s.items.forEach(function(it){L+=it.name+'\n  '+(Math.round(it.qty*100)/100)+' kg x '+num(it.price)+' = '+num(it.qty*it.price)+'\n'});
  var d=s.total-s.paid;
  L+='--------------------------\nTOTAL      KSh '+num(s.total)+'\n';
  L+='Paid ('+({cash:'cash',mpesa:'M-Pesa',credit:'credit',other:'other'}[s.method]||'cash')+') KSh '+num(s.paid)+'\n';
  if(d>0.5)L+='BALANCE    KSh '+num(d)+'\n';
  if(s.ref)L+='M-Pesa code: '+s.ref+'\n';
  return L+'Thank you for your business!';
}
function receipt(id){
  var s=S.sales.find(function(x){return x.id===id});if(!s)return;
  var gp=s.total-(ENG.cogs[s.id]||0);
  sheet(sheetHead('Receipt')+'<pre class="receipt" id="rcpt">'+esc(receiptText(s))+'</pre>'+
   '<div class="small muted">Profit on this sale: <b class="good">'+money(gp)+'</b> ('+pct(gp,s.total)+')</div>'+
   '<div class="actions"><button class="btn" data-act="copy" data-id="'+s.id+'">Copy receipt</button><button class="btn ghost" data-act="close">Done</button></div>'+
   '<button class="btn danger" data-act="delSale" data-id="'+s.id+'">Delete this sale</button>');
}
function copyText(t,el){
  var fb=function(){try{var r=document.createRange();r.selectNodeContents(el);var sel=getSelection();sel.removeAllRanges();sel.addRange(r);toast('Receipt selected. Press and hold to copy.')}catch(e){}};

}

  var rows=[['Date','Receipt','Customer','Items','Total','Paid','Method','M-Pesa code','Cost','Profit']];
  S.sales.slice().sort(function(a,b){return a.ts-b.ts}).forEach(function(s){var c=cust(s.cid),cost=ENG.cogs[s.id]||0;rows.push([new Date(s.ts).toISOString().slice(0,16).replace('T',' '),s.no,c?c.name:'Walk-in',s.items.map(function(i){return i.name+' '+i.qty+'kg@'+i.price}).join('; '),s.total,s.paid,s.method,s.ref||'',Math.round(cost),Math.round(s.total-cost)])});
  if(rows.length<2)return toast('No sales to export yet.');
  var csv=rows.map(function(r){return r.map(q).join(',')}).join('\n');

}

async function loadSample(){
  if(S.products.length&&!confirmSample())return;
  toast('Adding sample data…');
  var now=Date.now(),seed=7,rand=function(){seed=(seed*16807)%2147483647;return seed/2147483647};
  var mk=function(c,o){o.sample=true;return put(c,o,{quiet:true})};
  var defs=[['Sample item A',650],['Sample item B',600],['Sample item C',700],['Sample item D',580],['Sample item E',300],['Sample item F',420]],ps=[];
  for(var i=0;i<defs.length;i++)ps.push(await mk('products',{name:defs[i][0],price:defs[i][1],reorder:5,ts:now}));
  for(i=0;i<ps.length;i++){await mk('purchases',{pid:ps[i].id,qty:40,cost:Math.round(40*defs[i][1]*0.68),note:'Opening stock',ts:now-6.6*DAY});await mk('purchases',{pid:ps[i].id,qty:20,cost:Math.round(20*defs[i][1]*0.73),note:'Top-up',ts:now-3*DAY})}
  var cs=[];for(i=0;i<3;i++)cs.push(await mk('customers',{name:['Mama Njeri','Kioko Eatery','Baraka Grill'][i],phone:'',ts:now}));
  var k=0;
  for(var d=6;d>=0;d--){for(var n=0;n<2;n++){
    var ts=now-d*DAY-(n+1)*2.5*3600e3,items=[],m=1+Math.floor(rand()*2.99);
    for(var j=0;j<m;j++){var p=ps[Math.floor(rand()*ps.length)];if(items.some(function(x){return x.pid===p.id}))continue;items.push({pid:p.id,name:p.name,qty:rnd2(1+Math.round(rand()*50)/10),price:p.price})}
    var total=rnd2(sum(items,function(x){return x.qty*x.price})),useC=rand()<0.35,cu=useC?cs[Math.floor(rand()*3)]:null,meth=rand()<0.5?'mpesa':'cash',paid=total;
    if(cu&&rand()<0.6){paid=0;meth='credit'}
    k++;
    await mk('sales',{ts:ts,no:'PP-SAMPLE-'+String(k).padStart(3,'0'),cid:cu?cu.id:null,items:items,total:total,paid:paid,method:meth,ref:meth==='mpesa'?'SJ'+Math.floor(rand()*1e8).toString(36).toUpperCase().padEnd(8,'X'):''});
  }}
  await mk('payments',{cid:cs[0].id,amount:1000,method:'mpesa',ref:'SJ4K9P2LQ7',ts:now-1*DAY});
  await mk('expenses',{cat:'Transport',amount:800,note:'Pig collection',ts:now-5*DAY});
  await mk('expenses',{cat:'Electricity',amount:1500,note:'Cold room tokens',ts:now-4*DAY});
  await mk('expenses',{cat:'Ice / cold storage',amount:600,note:'',ts:now-1*DAY});
  render();toast('Sample data added. Remove it any time from More.');
}
